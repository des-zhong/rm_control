cmake_minimum_required(VERSION 3.16)
project(as2_follow_agent LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT UNITREE_SDK_DIR)
  message(FATAL_ERROR "Pass -DUNITREE_SDK_DIR=/absolute/path/to/unitree_sdk2")
endif()
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${UNITREE_SDK_DIR} sdk EXCLUDE_FROM_ALL)
add_executable(as2_agent agent.cpp)
target_link_libraries(as2_agent PRIVATE unitree_sdk2)
